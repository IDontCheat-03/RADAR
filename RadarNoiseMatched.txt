clear;
close all;

%% =====================================================
%        RADAR + TARGET + NOISE (PULSE RADAR)
%        Simple, Syllabus-Level, Error-Free
%% =====================================================

%% ---------------- Radar Parameters ----------------
c = 3e8;              % Speed of light (m/s)
fc = 10e9;            % Radar frequency (10 GHz)
lambda = c/fc;        % Wavelength (m)

Pt = 1e3;             % Transmit power (W)
G = 30;               % Antenna gain (linear)
sigma = 1;            % Radar Cross Section (m^2)

%% ---------------- Target ----------------
R = 3000;             % Target range (meters)

%% ---------------- Pulse ----------------
Tp = 1e-6;            % Pulse width (1 microsecond)
fs = 10e6;            % Sampling frequency
t = 0:1/fs:50e-6;     % Time window

%% ---------------- Transmitted Pulse ----------------
tx = rectpuls(t,Tp); % Rectangular pulse

%% ---------------- Radar Equation ----------------
Pr = (Pt * G^2 * lambda^2 * sigma) / ((4*pi)^3 * R^4);

%% ---------------- Echo Generation ----------------
delay = 2*R/c;                      % Round-trip delay
delay_samples = round(delay*fs);    % Delay in samples

rx = zeros(size(tx));               % Initialize received signal
rx(delay_samples+1 : delay_samples+length(tx)) = ...
    sqrt(Pr) * tx;                  % Delayed echo

%% ---------------- Receiver Noise ----------------
noise_power = 1e-12;                % Noise power
noise = sqrt(noise_power/2)*randn(size(rx));

rx_noisy = rx + noise;              % Signal + noise

%% ---------------- Matched Filter ----------------
mf = fliplr(tx);                    % Time-reversed pulse
mf_output = conv(rx_noisy, mf);     % Matched filter output

%% ---------------- Plots (NO SIZE MISMATCH) ----------------
figure('Color','w')

subplot(3,1,1)
plot(tx, 'LineWidth', 1.5)
title('Transmitted Pulse')
xlabel('Sample Index')
ylabel('Amplitude')
grid on

subplot(3,1,2)
plot(rx_noisy, 'LineWidth', 1)
title('Received Signal (Target + Noise)')
xlabel('Sample Index')
ylabel('Amplitude')
grid on

subplot(3,1,3)
plot(mf_output, 'LineWidth', 1.5)
title('Matched Filter Output (Detection)')
xlabel('Sample Index')
ylabel('Amplitude')
grid on
