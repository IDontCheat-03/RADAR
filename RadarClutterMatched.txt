clc;
clear;
close all;

%% =====================================================
%        RADAR + TARGET + CLUTTER (PULSE RADAR)
%        SIMPLE, ERROR-FREE
%% =====================================================

%% ---------------- Radar Parameters ----------------
c = 3e8;              % Speed of light (m/s)
fc = 10e9;            % Radar frequency (10 GHz)
lambda = c/fc;        % Wavelength (m)

Pt = 1e3;             % Transmit power (W)
G = 30;               % Antenna gain (linear)
sigma = 1;            % Radar Cross Section (m^2)

%% ---------------- Target ----------------
R = 3000;             % Target range (meters)

%% ---------------- Pulse ----------------
Tp = 1e-6;            % Pulse width
fs = 10e6;            % Sampling frequency
t = 0:1/fs:50e-6;     % Time window
N = length(t);        % Total samples

%% ---------------- Transmitted Pulse ----------------
tx = rectpuls(t, Tp);

%% ---------------- Radar Equation ----------------
Pr = (Pt * G^2 * lambda^2 * sigma) / ((4*pi)^3 * R^4);

%% ---------------- Target Echo ----------------
delay = 2*R/c;                      % Round-trip delay
delay_samples = round(delay * fs);  % Delay in samples

rx_target = zeros(1, N);            % Initialize received signal

% Safe echo insertion (NO size mismatch)
for n = 1:length(tx)
    idx = n + delay_samples;
    if idx <= N
        rx_target(idx) = sqrt(Pr) * tx(n);
    end
end

%% ---------------- Clutter Generation ----------------
% Simple clutter: low-level random reflections
clutter_power = 0.01 * Pr;           % Clutter weaker than target
clutter = sqrt(clutter_power) * rand(1, N);

%% ---------------- Received Signal ----------------
rx = rx_target + clutter;            % Target + clutter ONLY

%% ---------------- Matched Filter ----------------
mf = fliplr(tx);                     % Time-reversed pulse
mf_output = conv(rx, mf);

%% ---------------- Plots ----------------
figure('Color','w')

subplot(3,1,1)
plot(tx, 'LineWidth', 1.5)
title('Transmitted Pulse')
xlabel('Sample Index')
ylabel('Amplitude')
grid on

subplot(3,1,2)
plot(rx, 'LineWidth', 1)
title('Received Signal (Target + Clutter)')
xlabel('Sample Index')
ylabel('Amplitude')
grid on

subplot(3,1,3)
plot(mf_output, 'LineWidth', 1.5)
title('Matched Filter Output')
xlabel('Sample Index')
ylabel('Amplitude')
grid on
