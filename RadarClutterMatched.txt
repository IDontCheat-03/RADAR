clc;
clear;
close all;

%% ================= Radar Parameters =================
c = 3e8;              % Speed of light (m/s)
fc = 10e9;            % Carrier frequency (Hz)
lambda = c/fc;        % Wavelength (m)

Pt = 1e3;             % Transmit power (W)
G  = 30;              % Antenna gain (linear)
sigma = 1;            % RCS (m^2)

%% ================= Target =================
R = 3000;             % Target range (m)

%% ================= Pulse =================
Tp = 1e-6;            % Pulse width (s)
fs = 10e6;            % Sampling frequency (Hz)
t  = 0:1/fs:50e-6;    % Time axis
N  = length(t);

%% ================= Transmitted Pulse =================
tx = rectpuls(t - Tp/2, Tp);   % Centered rectangular pulse

%% ================= Radar Equation =================
Pr = (Pt * G^2 * lambda^2 * sigma) / ((4*pi)^3 * R^4);

%% ================= Target Echo =================
delay = 2*R/c;                    % Round-trip delay
delay_samples = round(delay*fs);  % Delay in samples

rx_target = zeros(1, N);

for n = 1:length(tx)
    idx = n + delay_samples;
    if idx <= N
        rx_target(idx) = sqrt(Pr) * tx(n);
    end
end

%% ================= Rayleigh Clutter (INBUILT) =================
clutter_power = 0.1 * Pr;                % Clutter weaker than target
clutter = raylrnd(sqrt(clutter_power), 1, N);

%% ================= Received Signal =================
rx = rx_target + clutter;

%% ================= Matched Filter =================
mf = fliplr(tx);                  % Time-reversed pulse
mf_out = conv(rx, mf, 'same');    % Same-length output

%% ================= Plots =================
figure('Color','w')

subplot(3,1,1)
plot(t*1e6, tx, 'LineWidth', 1.5)
title('Transmitted Pulse')
xlabel('Time (\mus)')
ylabel('Amplitude')
grid on

subplot(3,1,2)
plot(t*1e6, rx)
title('Received Signal (Target + Rayleigh Clutter)')
xlabel('Time (\mus)')
ylabel('Amplitude')
grid on

subplot(3,1,3)
plot(t*1e6, mf_out, 'LineWidth', 1.5)
title('Matched Filter Output')
xlabel('Time (\mus)')
ylabel('Amplitude')
grid on
